This lab used to build a Django template out of your post and write it into the template directory, so the next page load compiled whatever you typed: a {% ... %} tag ran on the server. The post is now stored as data and handed to one fixed template as context, so the engine escapes it. Compare the two versions below - the difference is template source versus template context.

def ssti_lab(request):
 if request.user.is_authenticated:
  if request.method=="GET":
   users_blogs = Blogs.objects.filter(author=request.user)
   return render(request,"Lab_2021/A3_Injection/ssti_lab.html", {"blogs":users_blogs})
  elif request.method=="POST":
   blog = request.POST["blog"]
   id = str(uuid.uuid4()).split('-')[-1]
   
   # vulnerable version: the post became template source on disk
   # blog = prepend_code + blog + "{% endblock %}"
   # open(f"templates/Lab_2021/A3_Injection/Blogs/{id}.html", "w+").write(blog)
   
   # fixed version: the post is stored as data
   new_blog = Blogs.objects.create(author=request.user, blog_id=id, content=blog)
   new_blog.save()
   return redirect(f'blog/{id}')
 else:
  return redirect('login')

def ssti_view_blog(request, blog_id):
 if request.user.is_authenticated:
  if request.method=="GET":
   blog = Blogs.objects.filter(blog_id=blog_id).first()
   if blog is None:
    return HttpResponseBadRequest()
   # one fixed template, the body passed as autoescaped context
   return render(request, "Lab_2021/A3_Injection/ssti_blog.html", {"blog": blog})
