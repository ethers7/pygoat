def ssti_lab(request):
 if request.user.is_authenticated:
  if request.method=="GET":
   users_blogs = Blogs.objects.filter(author=request.user)
   return render(request,"Lab_2021/A3_Injection/ssti_lab.html", {"blogs":users_blogs})
  elif request.method=="POST":
   # oversize bodies are rejected with a 400 instead of being written
   blog = validate_submitted_text(request.POST.get("blog"), MAX_BLOG_BODY_BYTES)
   blog = filter_blog(blog) # strips template syntax
   id = str(uuid.uuid4()).split('-')[-1]
   # fixed directory outside the template search path, path checked
   filename = resolve_inside_directory(SSTI_BLOG_DIR, f"{id}.txt")
   write_text_exclusively(filename, blog) # data file, never a template
   new_blog = Blogs.objects.create(author = request.user, blog_id = id)
   new_blog.save()
   return redirect(f'blog/{id}')
  # the saved body is printed through an autoescaped context variable,
  # so it is displayed as data and never evaluated server side
 else:
  return redirect('login')
