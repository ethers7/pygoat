Inspect the buttons and edit the hidden "blog" field: try to walk out of the blogs directory (../../.env, ../../pygoat/settings.py) or to name a file outright (/etc/passwd). The view resolves the name inside the blogs directory only, so every one of those is refused with "Refused: only the blogs of this lab can be read" - compare that with the vulnerable version in the discussion page, which happily returned the .env file.
# utility.py - the request only ever names a blog, never a path
_BLOG_ROOT = os.path.realpath(os.path.join(os.path.dirname(os.path.realpath(__file__)),
           'templates', 'Lab', 'ssrf', 'blogs'))
_BLOG_NAME_RE = re.compile(r'\A[A-Za-z0-9][A-Za-z0-9_-]{0,63}\.txt\Z')

def blog_path(name):
 if not isinstance(name, str) or not _BLOG_NAME_RE.match(name):
  raise BlogNotAllowed('a blog is named like "blog1.txt"') # no separator, no .., no absolute path
 path = os.path.realpath(os.path.join(_BLOG_ROOT, name)) # resolves symlinks too
 if os.path.dirname(path) != _BLOG_ROOT:
  raise BlogNotAllowed('blog is outside the blogs directory')
 if not os.path.isfile(path):
  raise BlogNotAllowed('no such blog')
 return path

# views.py
def ssrf_lab(request):
 if request.user.is_authenticated:
  if request.method=="GET":
   return render(request,"Lab/ssrf/ssrf_lab.html",{"blog":"Read Blog About SSRF"})
  else:
   try:
    filename = blog_path(request.POST.get("blog"))
   except BlogNotAllowed:
    return render(request, "Lab/ssrf/ssrf_lab.html", {"blog": "Refused: ..."}, status=400)
   try:
    with open(filename, "r", encoding="utf-8") as blog_file:
     data = blog_file.read()
   except (OSError, UnicodeDecodeError):
    return render(request, "Lab/ssrf/ssrf_lab.html", {"blog": "No blog found"}, status=404)
   return render(request, "Lab/ssrf/ssrf_lab.html", {"blog": data})
 else:
  return redirect('login')
