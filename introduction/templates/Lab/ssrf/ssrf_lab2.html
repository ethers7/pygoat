@authentication_decorator
def ssrf_lab2(request):
 if request.method == "GET":
  return render(request, "Lab/ssrf/ssrf_lab2.html")

 elif request.method == "POST":
  url = request.POST.get("url", "")
  try:
   response = fetch_url_safely(url) # validates scheme/host, blocks internal + metadata IPs, re-checks redirects
  except UnsafeUrlError as exc:
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": str(exc)})
  except requests.RequestException:
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": "Invalid URL"})
  return render(request, "Lab/ssrf/ssrf_lab2.html", {"response": response.content.decode(errors="replace")})
