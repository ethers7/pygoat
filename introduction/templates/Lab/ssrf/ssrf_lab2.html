@authentication_decorator
def ssrf_lab2(request):
 allowed_hosts = sorted(fetch_allowed_hosts())
 if request.method == "GET":
  return render(request, "Lab/ssrf/ssrf_lab2.html", {"allowed_hosts": allowed_hosts})

 elif request.method == "POST":
  # only fetch an allowlisted http(s) host that resolves to a public address
  target = safe_fetch_url(request.POST.get("url"))
  if target is None:
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": "URL not allowed. ..."})
  try:
   # redirects are re-validated with safe_fetch_url, never followed blindly
   # and the fetch is bounded in time and size, never unlimited
   response = requests.get(target, timeout=FETCH_TIMEOUT, allow_redirects=False, stream=True)
   body = read_bounded_response(response)  # at most MAX_FETCH_RESPONSE_BYTES
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"response": body.decode(errors="replace")})
  except ResponseTooLargeError:
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": "Response too large"})
  except requests.RequestException:
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": "Invalid URL"})
