@authentication_decorator
def ssrf_lab2(request):
 if request.method == "GET":
  return render(request, "Lab/ssrf/ssrf_lab2.html")

 elif request.method == "POST":
  target = safe_outbound_url(request.POST["url"]) # http(s) + host allowlist + public address only
  if target is None:
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": SSRF_LAB_REJECTED})
  try:
   for _ in range(SSRF_LAB_MAX_REDIRECTS + 1): # bounded redirect chain
    with requests.get(target, timeout=OUTBOUND_FETCH_TIMEOUT, allow_redirects=False, stream=True) as response: # (connect, read) timeout
     if not response.is_redirect:
      body = read_bounded_body(response) # at most MAX_OUTBOUND_RESPONSE_BYTES
      if body is None:
       return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": SSRF_LAB_TOO_LARGE})
      return render(request, "Lab/ssrf/ssrf_lab2.html", {"response": body.decode(errors="replace")})
     location = response.headers.get("location", "")
    target = safe_outbound_url(urljoin(target, location)) # re-validate each hop
    if target is None:
     return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": SSRF_LAB_REJECTED})
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": "Too many redirects"})
  except requests.RequestException:
   return render(request, "Lab/ssrf/ssrf_lab2.html", {"error": "Invalid URL"})
